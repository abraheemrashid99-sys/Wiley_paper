\node[term] (start) {Step $n$: messages $\mathbf{x}_1,\dots,\mathbf{x}_{N_\mathrm{c}}$, SNR $\gamma_n$};

\node[txproc, below=of d1] (enc) {Encode every token; send round 1 ($m_{k_0} = 6$ symbols per token)};

\node[rxproc, below=of d2] (ctx) {Deliver the message; append it to the context $\mathbf{h}$ (fast memory)};
\node[txproc, below=of ctx] (learn) {One optimizer step on $\mathcal{L}_n$ with the true text};
\node[dec, below=of learn] (d3) {$n \bmod C_\ell = 0$ for a level $\ell$?};
\node[side, right=12mm of d3] (ship) {Ship $\mathcal{Q}_b(\Delta_\ell)$, set both copies equal and record $B_\ell$};
